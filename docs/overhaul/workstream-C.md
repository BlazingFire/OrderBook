**Model.** Split the two jobs `[data-density]` conflates today. `--root-size` becomes the continuous knob; `[data-density]` keeps only the height-to-text *ratios*, in rem.

**Persistence.** Extend `master/web/src/ui/theme.ts` and `office/web/src/ui/theme.ts` **by hand, separately** — they are forks. Add to each, changing nothing else:

```ts
const KEY_SCALE = 'ob.scale';
export const SCALE_MIN = 11, SCALE_MAX = 22, SCALE_STEP = 0.5, SCALE_DEFAULT = 14;
const clamp = (n: number) => Math.min(SCALE_MAX, Math.max(SCALE_MIN, n));

export function loadScale(): number {
  try {
    const raw = window.localStorage.getItem(KEY_SCALE);
    if (raw !== null) { const n = Number(raw); if (Number.isFinite(n)) return clamp(n); }
    return { compact: 13, comfortable: 14, large: 16 }[loadDensity()];  // one-time migration
  } catch { return SCALE_DEFAULT; }
}

export function applyScale(px: number): void {
  const v = clamp(px);
  document.documentElement.style.setProperty('--root-size', `${v}px`);
  store(KEY_SCALE, String(v));
}
```

### 2. Responsiveness

`minWidth: 1366` → `1024` in both shells; `minHeight: 768` and the 1440x900 default unchanged.
